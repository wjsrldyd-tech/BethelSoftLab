# 대항해시대 오리진 계절 시스템

## 개요

대항해시대 오리진의 계절은 **7가지 타입**으로 분류됩니다.

- **월 전환 시간**: 매일 아침 9시 (KST)
- **특별 규칙**: 
  - 열대, 한대는 별도의 성수기가 없습니다
  - 식료품 중 '밀'은 성수기/비수기가 별도로 존재하지 않습니다

## 구현 참고 사항

### 계절 축(Season Axis) 결정
각 항구의 계절 시스템은 다음과 같이 결정됩니다:

1. **season 축 (사계절)**: 북반구-사계절, 남반구-사계절
2. **climate 축 (건기/우기)**: 건기-우기-건기, 건기+1-우기-건기, 우기-건기-우기
3. **none**: 열대, 한대

### 성수기/비수기 판정 로직

1. 현재 월과 항구의 계절 타입으로 현재 계절/기후를 결정
2. 교역품의 카테고리 확인
3. 해당 계절 타입과 카테고리에 따른 성수기/비수기 규칙 적용
4. 성수기: 수량 × 1.5, 비수기: 수량 × 0.5, 평수기: 수량 × 1.0

### 코드 구조

```javascript
// origin-ports-data.js

// 항구 → 계절 타입 매핑 (170개 항구)
ORIGIN_PORT_SEASON_TYPE = {
  '낭트': 'north-4seasons',
  '리마': 'south-4seasons',
  '마닐라': 'dry-rainy-dry',
  '하바나': 'dry+1-rainy-dry',
  '다바오': 'tropical',
  '레이캬비크': 'arctic',
  // ... (170개 항구 전체 매핑)
}

window.getOriginPortSeasonType = function(portName) {
  return ORIGIN_PORT_SEASON_TYPE[portName] || 'north-4seasons';
}

// origin-port-goods.js

// 계절 타입별 월 → 태그 매핑 (배열 인덱스 0~11 = 월 1~12)
SEASON_TYPE_CALENDARS = {
  'north-4seasons': ['겨울', '겨울', '봄', '봄', '봄', '여름', 
                     '여름', '여름', '가을', '가을', '가을', '겨울'],
  'south-4seasons': ['여름', '여름', '가을', '가을', '가을', '겨울',
                     '겨울', '겨울', '봄', '봄', '봄', '여름'],
  'dry-rainy-dry': ['건기', '건기', '건기', '건기', '건기', '우기',
                    '우기', '우기', '우기', '우기', '건기', '건기'],
  'dry+1-rainy-dry': ['건기', '건기', '건기', '건기', '건기', '건기',
                      '우기', '우기', '우기', '우기', '우기', '건기'],
  'rainy-dry-rainy': ['우기', '우기', '우기', '우기', '우기', '건기',
                      '건기', '건기', '건기', '건기', '건기', '우기'],
  'tropical': ['', '', '', '', '', '', '', '', '', '', '', ''],
  'arctic': ['', '', '', '', '', '', '', '', '', '', '', '']
}

// 계절 타입 + 분류 → 성수기/비수기 매핑 (통합표 기준으로 추후 완성)
CATEGORY_SEASON_RULES = {
  'north-4seasons': {
    // '식료품': { peak: ['봄'], off: ['가을'] },
    // '주류': { peak: ['겨울'], off: ['여름'] },
    // 통합표 기준으로 추후 추가
  },
  'south-4seasons': { /* 남반구 규칙 */ },
  'dry-rainy-dry': { /* 건기-우기 규칙 */ },
  'dry+1-rainy-dry': { /* 건기+1-우기 규칙 */ },
  'rainy-dry-rainy': { /* 우기-건기 규칙 */ },
  'tropical': { /* 빈 객체 */ },
  'arctic': { /* 빈 객체 */ }
}

// 특수 예외 처리
ALWAYS_PLAIN_GOODS = ['밀'];

// 사용 예시
window.getOriginSeasonForMonth = function(month, portName) {
  const seasonType = getOriginPortSeasonType(portName);
  const calendar = SEASON_TYPE_CALENDARS[seasonType];
  const activeTag = calendar[month - 1];
  return { month, seasonType, activeTag };
}

window.getOriginGoodSeasonStatus = function(good, month, portName) {
  const { seasonType, activeTag } = getOriginSeasonForMonth(month, portName);
  
  // 1. 열대/한대 → 항상 plain
  if (seasonType === 'tropical' || seasonType === 'arctic') return 'plain';
  
  // 2. 밀 → 항상 plain
  if (good.name === '밀') return 'plain';
  
  // 3. 분류별 규칙 조회
  const categoryRule = CATEGORY_SEASON_RULES[seasonType][good.category];
  if (categoryRule) {
    if (categoryRule.peak.includes(activeTag)) return 'peak';
    if (categoryRule.off.includes(activeTag)) return 'off';
  }
  
  // 4. 품목별 태그로 폴백 (호환성)
  if (good.peak && good.peak.includes(activeTag)) return 'peak';
  if (good.off && good.off.includes(activeTag)) return 'off';
  
  return 'plain';
}
```
